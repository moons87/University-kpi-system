# ETL Pipeline Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Pandas-based ETL pipeline that reads data from PostgreSQL, calculates KPI metrics, writes results back to the database, and exports CSV/Excel files for Power BI.

**Architecture:** Single orchestrator script (`pipeline.py`) with two transformer modules. `kpi_aggregator.py` mirrors the backend KPI engine using pure Pandas (no SQLAlchemy ORM — raw SQL via `pd.read_sql`). `analytics_builder.py` builds summary tables for Power BI export. Outputs go to `etl/exports/`.

**Tech Stack:** Python 3.11+, Pandas, SQLAlchemy (engine only), openpyxl, psycopg2-binary, python-dotenv

**Prerequisite:** Plan 1 complete (database with seed data), Plan 2 complete (kpi_scores and kpi_details tables exist).

### Task 2: KPI aggregator transformer

**Files:**
- Create: `etl/transformers/kpi_aggregator.py`

- [ ] **Step 1: Create etl/transformers/kpi_aggregator.py**

```python
import pandas as pd
from sqlalchemy import text
from sqlalchemy.engine import Engine

# KPI weights (must sum to 1.0)
WEIGHTS = {
    "teaching":    0.30,
    "research":    0.35,
    "project":     0.15,
    "achievement": 0.20,
}

MAX_VALUES = {
    "hours_total":          300,
    "scopus_wos_count":     5,
    "local_count":          10,
    "patent_count":         3,
    "project_count":        3,
    "project_budget":       10_000_000,
    "achievement_intl":     2,
    "achievement_natl":     4,
    "achievement_local":    6,
}


def _norm(value: float, max_val: float) -> float:
    return round(min(float(value), max_val) / max_val * 100, 2) if max_val > 0 else 0.0


def run(engine: Engine, year: int, semester: int) -> dict:
    """
    Calculate KPI for all teachers for the given period.
    Writes to kpi_scores and kpi_details tables.
    Returns {"calculated": N}.
    """
    with engine.connect() as conn:
        # Resolve time_id
        time_row = conn.execute(
            text("SELECT id FROM time_dim WHERE year=:y AND semester=:s"),
            {"y": year, "s": semester}
        ).fetchone()
        if not time_row:
            raise ValueError(f"No time_dim entry for year={year} semester={semester}")
        time_id = time_row[0]

        teachers_df = pd.read_sql(
            "SELECT id, full_name FROM teachers", conn
        )
        tl_df = pd.read_sql(
            f"SELECT teacher_id, hours FROM teaching_load WHERE time_id={time_id}", conn
        )
        pub_df = pd.read_sql(
            f"SELECT teacher_id, type FROM publications WHERE time_id={time_id}", conn
        )
        pat_df = pd.read_sql(
            f"SELECT teacher_id FROM patents WHERE time_id={time_id}", conn
        )
        ach_df = pd.read_sql(
            f"SELECT teacher_id, level FROM achievements WHERE time_id={time_id}", conn
        )
        proj_df = pd.read_sql(
            f"SELECT teacher_id, COALESCE(budget, 0) AS budget FROM projects WHERE time_id={time_id}", conn
        )

        # Clear existing results
        conn.execute(text(f"DELETE FROM kpi_details WHERE time_id={time_id}"))
        conn.execute(text(f"DELETE FROM kpi_scores  WHERE time_id={time_id}"))
        conn.commit()

    scores_rows  = []
    details_rows = []

    for _, teacher in teachers_df.iterrows():
        tid = int(teacher["id"])

        # --- Teaching ---
        hours = tl_df[tl_df.teacher_id == tid]["hours"].sum() if not tl_df.empty else 0
        teaching_score = _norm(hours, MAX_VALUES["hours_total"])
        details_rows.append({"teacher_id": tid, "time_id": time_id, "category": "teaching",
                              "metric_name": "hours_total", "value": float(hours), "score": teaching_score})

        # --- Research ---
        t_pub = pub_df[pub_df.teacher_id == tid] if not pub_df.empty else pd.DataFrame(columns=["type"])
        sw_count  = len(t_pub[t_pub["type"].isin(["Scopus", "WoS"])])
        loc_count = len(t_pub[t_pub["type"] == "local"])
        pat_count = len(pat_df[pat_df.teacher_id == tid]) if not pat_df.empty else 0

        sw_sc  = _norm(sw_count,  MAX_VALUES["scopus_wos_count"])
        loc_sc = _norm(loc_count, MAX_VALUES["local_count"])
        pat_sc = _norm(pat_count, MAX_VALUES["patent_count"])
        research_score = round(sw_sc * 0.6 + loc_sc * 0.2 + pat_sc * 0.2, 2)

        for metric, val, sc in [
            ("scopus_wos_count", sw_count,  sw_sc),
            ("local_pub_count",  loc_count, loc_sc),
            ("patent_count",     pat_count, pat_sc),
        ]:
            details_rows.append({"teacher_id": tid, "time_id": time_id, "category": "research",
                                  "metric_name": metric, "value": float(val), "score": sc})

        # --- Projects ---
        t_proj = proj_df[proj_df.teacher_id == tid] if not proj_df.empty else pd.DataFrame(columns=["budget"])
        p_count  = len(t_proj)
        p_budget = float(t_proj["budget"].sum()) if not t_proj.empty else 0
        pc_sc = _norm(p_count,  MAX_VALUES["project_count"])
        pb_sc = _norm(p_budget, MAX_VALUES["project_budget"])
        project_score = round(pc_sc * 0.5 + pb_sc * 0.5, 2)

        for metric, val, sc in [("project_count", p_count, pc_sc), ("project_budget", p_budget, pb_sc)]:
            details_rows.append({"teacher_id": tid, "time_id": time_id, "category": "project",
                                  "metric_name": metric, "value": float(val), "score": sc})

        # --- Achievements ---
        t_ach = ach_df[ach_df.teacher_id == tid] if not ach_df.empty else pd.DataFrame(columns=["level"])
        intl = len(t_ach[t_ach.level == "international"])
        natl = len(t_ach[t_ach.level == "national"])
        locl = len(t_ach[t_ach.level == "local"])
        intl_sc = _norm(intl, MAX_VALUES["achievement_intl"])
        natl_sc = _norm(natl, MAX_VALUES["achievement_natl"])
        locl_sc = _norm(locl, MAX_VALUES["achievement_local"])
        achievement_score = round(intl_sc * 0.5 + natl_sc * 0.35 + locl_sc * 0.15, 2)

        for metric, val, sc in [
            ("achievement_international", intl, intl_sc),
            ("achievement_national",      natl, natl_sc),
            ("achievement_local",         locl, locl_sc),
        ]:
            details_rows.append({"teacher_id": tid, "time_id": time_id, "category": "achievement",
                                  "metric_name": metric, "value": float(val), "score": sc})

        total_score = round(
            teaching_score    * WEIGHTS["teaching"]    +
            research_score    * WEIGHTS["research"]    +
            project_score     * WEIGHTS["project"]     +
            achievement_score * WEIGHTS["achievement"],
            2,
        )

        scores_rows.append({
            "teacher_id": tid, "time_id": time_id,
            "teaching_score": teaching_score, "research_score": research_score,
            "project_score": project_score,   "achievement_score": achievement_score,
            "total_score": total_score,
        })

    # Bulk insert
    scores_df  = pd.DataFrame(scores_rows)
    details_df = pd.DataFrame(details_rows)

    scores_df.to_sql("kpi_scores",  engine, if_exists="append", index=False)
    details_df.to_sql("kpi_details", engine, if_exists="append", index=False)

    print(f"[kpi_aggregator] Calculated KPI for {len(scores_rows)} teachers (year={year}, semester={semester})")
    return {"calculated": len(scores_rows)}
```

```python
import pandas as pd
from pathlib import Path
from sqlalchemy.engine import Engine

EXPORTS_DIR = Path(__file__).parent.parent / "exports"


def run(engine: Engine, year: int, semester: int) -> list[str]:
    """
    Build summary DataFrames and export to CSV and Excel.
    Returns list of exported file paths.
    """
    EXPORTS_DIR.mkdir(exist_ok=True)
    exported = []

    with engine.connect() as conn:
        # 1. KPI summary per teacher
        kpi_df = pd.read_sql(f"""
            SELECT
                t.full_name,
                d.name  AS department,
                p.name  AS position,
                k.teaching_score,
                k.research_score,
                k.project_score,
                k.achievement_score,
                k.total_score
            FROM kpi_scores k
            JOIN teachers   t ON t.id = k.teacher_id
            LEFT JOIN departments d ON d.id = t.department_id
            LEFT JOIN positions   p ON p.id = t.position_id
            JOIN time_dim td ON td.id = k.time_id
            WHERE td.year = {year} AND td.semester = {semester}
            ORDER BY k.total_score DESC
        """, conn)

        # 2. Publication counts per teacher
        pub_df = pd.read_sql(f"""
            SELECT
                t.full_name,
                p.type,
                p.quartile,
                COUNT(*) AS count
            FROM publications p
            JOIN teachers t ON t.id = p.teacher_id
            JOIN time_dim td ON td.id = p.time_id
            WHERE td.year = {year} AND td.semester = {semester}
            GROUP BY t.full_name, p.type, p.quartile
            ORDER BY t.full_name
        """, conn)

        # 3. Teaching hours per teacher
        tl_df = pd.read_sql(f"""
            SELECT
                t.full_name,
                s.name AS subject,
                g.name AS group_name,
                tl.hours
            FROM teaching_load tl
            JOIN teachers t ON t.id = tl.teacher_id
            JOIN subjects s ON s.id = tl.subject_id
            JOIN groups   g ON g.id = tl.group_id
            JOIN time_dim td ON td.id = tl.time_id
            WHERE td.year = {year} AND td.semester = {semester}
            ORDER BY t.full_name
        """, conn)

        # 4. Projects per teacher
        proj_df = pd.read_sql(f"""
            SELECT
                t.full_name,
                pr.title,
                pr.funding_source,
                pr.budget
            FROM projects pr
            JOIN teachers t ON t.id = pr.teacher_id
            JOIN time_dim td ON td.id = pr.time_id
            WHERE td.year = {year} AND td.semester = {semester}
            ORDER BY t.full_name
        """, conn)

    period_label = f"{year}_s{semester}"

    # Export CSV files
    kpi_csv  = EXPORTS_DIR / f"kpi_summary_{period_label}.csv"
    pub_csv  = EXPORTS_DIR / f"publications_{period_label}.csv"
    tl_csv   = EXPORTS_DIR / f"teaching_load_{period_label}.csv"
    proj_csv = EXPORTS_DIR / f"projects_{period_label}.csv"

    kpi_df.to_csv(kpi_csv,   index=False)
    pub_df.to_csv(pub_csv,   index=False)
    tl_df.to_csv(tl_csv,    index=False)
    proj_df.to_csv(proj_csv, index=False)
    exported += [str(kpi_csv), str(pub_csv), str(tl_csv), str(proj_csv)]

    # Export Excel workbook (all sheets in one file)
    xlsx_path = EXPORTS_DIR / f"university_analytics_{period_label}.xlsx"
    with pd.ExcelWriter(xlsx_path, engine="openpyxl") as writer:
        kpi_df.to_excel(writer,  sheet_name="KPI Summary",    index=False)
        pub_df.to_excel(writer,  sheet_name="Publications",   index=False)
        tl_df.to_excel(writer,   sheet_name="Teaching Load",  index=False)
        proj_df.to_excel(writer, sheet_name="Projects",       index=False)
    exported.append(str(xlsx_path))

    print(f"[analytics_builder] Exported {len(exported)} files to {EXPORTS_DIR}")
    return exported
```

```python
"""
University Analytics ETL Pipeline
Usage: python pipeline.py --year 2024 --semester 1
"""
import argparse
import os
from dotenv import load_dotenv
from sqlalchemy import create_engine

from transformers import kpi_aggregator, analytics_builder

load_dotenv()


def main():
    parser = argparse.ArgumentParser(description="University Analytics ETL Pipeline")
    parser.add_argument("--year",     type=int, required=True, help="Academic year (e.g. 2024)")
    parser.add_argument("--semester", type=int, choices=[1, 2], required=True, help="Semester (1 or 2)")
    args = parser.parse_args()

    database_url = os.getenv("DATABASE_URL")
    if not database_url:
        raise RuntimeError("DATABASE_URL not set in environment")

    engine = create_engine(database_url)
    print(f"\n=== ETL Pipeline: year={args.year}, semester={args.semester} ===\n")

    # Step 1: Calculate and persist KPI
    result = kpi_aggregator.run(engine, args.year, args.semester)
    print(f"KPI calculated: {result}")

    # Step 2: Build analytics and export files
    files = analytics_builder.run(engine, args.year, args.semester)
    print(f"\nExported files:")
    for f in files:
        print(f"  {f}")

    print("\n=== Pipeline complete ===\n")


if __name__ == "__main__":
    main()
```
